cmake_minimum_required(VERSION 3.21)
cmake_policy(SET CMP0091 NEW)
if(CMAKE_HOST_APPLE)
  set(CMAKE_OSX_DEPLOYMENT_TARGET "14.0" CACHE STRING "Pianpian minimum macOS" FORCE)
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
project(pianpian-align LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}/bin")
add_executable(pianpian-align main.cpp)
target_include_directories(pianpian-align PRIVATE "${HEADERS}")
if(MSVC)
  target_compile_definitions(pianpian-align PRIVATE NOMINMAX)
  target_compile_options(pianpian-align PRIVATE /utf-8)
  target_link_options(pianpian-align PRIVATE "/MANIFEST:EMBED" "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/windows.manifest")
else()
  target_link_libraries(pianpian-align PRIVATE ${CMAKE_DL_LIBS})
endif()
enable_testing()
add_test(NAME alignment-help COMMAND pianpian-align --help)
add_executable(alignment-protocol-test protocol_test.cpp)
target_include_directories(alignment-protocol-test PRIVATE "${HEADERS}")
if(MSVC)
  target_compile_definitions(alignment-protocol-test PRIVATE NOMINMAX)
  target_compile_options(alignment-protocol-test PRIVATE /utf-8)
else()
  target_link_libraries(alignment-protocol-test PRIVATE ${CMAKE_DL_LIBS})
endif()
add_test(NAME alignment-protocol COMMAND alignment-protocol-test)
