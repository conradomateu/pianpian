cmake_minimum_required(VERSION 3.21)
cmake_policy(SET CMP0091 NEW)
if(CMAKE_HOST_APPLE)
  set(CMAKE_OSX_DEPLOYMENT_TARGET "14.0" CACHE STRING "Pianpian's minimum macOS version" FORCE)
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
project(pianpian-sensevoice LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}/bin")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(GGML_CCACHE OFF CACHE BOOL "" FORCE)
# The baseline chooses a separately built AVX2 sibling only after CPU/OS checks.
set(PIANPIAN_CPU_VARIANT "baseline" CACHE STRING "baseline or avx2")
foreach(feature SSE42 AVX AVX2 BMI2 FMA F16C)
  set(GGML_${feature} OFF CACHE BOOL "" FORCE)
endforeach()
if(PIANPIAN_CPU_VARIANT STREQUAL "avx2")
  if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "The AVX2 sibling is only built for Windows x64")
  endif()
  foreach(feature SSE42 AVX AVX2 BMI2 FMA F16C)
    set(GGML_${feature} ON CACHE BOOL "" FORCE)
  endforeach()
elseif(NOT PIANPIAN_CPU_VARIANT STREQUAL "baseline")
  message(FATAL_ERROR "Unknown SenseVoice CPU variant")
endif()
if(MSVC)
  add_compile_definitions(NOMINMAX _USE_MATH_DEFINES)
  add_compile_options(/utf-8)
endif()
add_subdirectory("${UPSTREAM}/llama/ggml" ggml)
find_package(Threads REQUIRED)
add_executable(pianpian-sensevoice "${CMAKE_CURRENT_BINARY_DIR}/funasr-sensevoice.cpp")
if(PIANPIAN_CPU_VARIANT STREQUAL "avx2")
  target_compile_definitions(pianpian-sensevoice PRIVATE PIANPIAN_AVX2)
endif()
target_include_directories(pianpian-sensevoice PRIVATE "${UPSTREAM}/funasr/funasr-common" "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(pianpian-sensevoice PRIVATE ggml Threads::Threads)
if(MSVC)
  target_link_options(pianpian-sensevoice PRIVATE "/MANIFEST:EMBED" "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/windows.manifest")
endif()
add_executable(sensevoice-timestamps-test timestamps_test.cpp)
add_executable(sensevoice-cpu-test cpu_dispatch_test.cpp)
# Assertions remain active in Release builds: these are executable checks, not benchmarks.
if(MSVC)
  target_compile_options(sensevoice-timestamps-test PRIVATE /UNDEBUG)
  target_compile_options(sensevoice-cpu-test PRIVATE /UNDEBUG /utf-8)
else()
  target_compile_options(sensevoice-timestamps-test PRIVATE -UNDEBUG)
  target_compile_options(sensevoice-cpu-test PRIVATE -UNDEBUG)
endif()
enable_testing()
add_test(NAME sensevoice-timestamps COMMAND sensevoice-timestamps-test)
add_test(NAME sensevoice-cpu-dispatch COMMAND sensevoice-cpu-test)
add_test(NAME sensevoice-build-integrity COMMAND "${PYTHON_FOR_BUILD}" "${CMAKE_CURRENT_SOURCE_DIR}/test_builder.py")
